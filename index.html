PyXtend v2.4.0

PyXtend is a lightweight library that adds runtime type validation and serialization to Python dataclasses. It generates validators from type hints, supports nested models, and produces JSON schemas automatically.

Installation

pip install pyxtend

Quick Start

from dataclasses import dataclass
from pyxtend import validates, serialize

@dataclass
class User:
    name: str
    email: str
    age: int

# Runtime validation on instantiation
user = User(name="Alice", email="alice@example.com", age=30)
validates(user)  # True — all fields pass type checks

# Automatic JSON serialization
json_str = serialize(user)
# {"name": "Alice", "email": "alice@example.com", "age": 30}

Custom Validators

from pyxtend import field_validator

@dataclass  
class Product:
    name: str
    price: float
    sku: str

    @field_validator("price")
    def check_price(cls, v):
        assert v > 0, "price must be positive"
        return v

    @field_validator("sku")
    def check_sku(cls, v):
        import re
        assert re.match(r"^[A-Z]{3}-\d{4}$", v), "SKU format is XXX-0000"
        return v

Nested Models

from dataclasses import dataclass, field
from pyxtend import validates

@dataclass
class Address:
    street: str
    city: str
    zip: str

@dataclass
class Employee:
    name: str
    address: Address  # nested model — validated recursively
    tags: list[str] = field(default_factory=list)

emp = Employee(
    name="Bob",
    address=Address(street="123 Main St", city="Portland", zip="97201"),
    tags=["engineering", "senior"]
)
validates(emp)  # True — nested validation passes

Error Handling

from pyxtend import ValidationError, validates

try:
    user = User(name=123, email="not-an-email", age=-5)
    validates(user)
except ValidationError as e:
    print(f"Validation failed: {e}")
    # Validation failed: 3 errors:
    #   - name: expected str, got int
    #   - email: invalid email format
    #   - age: expected positive int, got -5

Migration from Pydantic

Most Pydantic models can be converted by changing the base class:

# Pydantic
from pydantic import BaseModel
class User(BaseModel):
    name: str

# PyXtend  
from dataclasses import dataclass
@dataclass
class User:
    name: str
